\section{Preferences}

\para A preference relation $\weakpref$ on a choice set $X$ is a binary
relation read as ``at least as good as''. \memo{add the axioms table here}

\para Two derived relations follow. Strict preference $\strictpref$ holds when
$x \weakpref y$ but not $y \weakpref x$; indifference $\indiff$ holds when
both directions hold.

\begin{definition}[Rational preferences]
A preference relation $\weakpref$ is \lapis{rational} if it is complete and
transitive.
\end{definition}

\begin{proposition}
If $\weakpref$ is rational then $\strictpref$ is irreflexive and transitive,
and $\indiff$ is an equivalence relation.
\end{proposition}

\begin{proof}
Immediate from completeness and transitivity. \qedhere
\end{proof}

\begin{notebox}
\boxentry{Why continuity matters}
Rationality alone does not deliver a utility representation. Continuity rules
out the lexicographic counterexample, and with it the jump that breaks any
candidate representation.
\end{notebox}

\section{Utility representation}

\para A function $u : X \to \R$ represents $\weakpref$ when
$x \weakpref y \iff u(x) \ge u(y)$.

\begin{theorem}[Debreu]
A rational and continuous preference relation on a connected subset of
$\R^n_+$ admits a continuous utility representation.
\end{theorem}

\lapisboxeq{x \weakpref y \iff u(x) \ge u(y)}

\begin{remark}
The representation is unique only up to a strictly increasing transformation.
\end{remark}

\begin{notebox}
\boxentry{Ordinal, not cardinal}
Nothing in the representation licenses comparing utility differences across
people, or even across bundles for one person.
\end{notebox}
