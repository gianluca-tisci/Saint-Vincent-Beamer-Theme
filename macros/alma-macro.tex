\newcommand{\Ct}{C_{t}}
\newcommand{\It}{I_{t}}

\newcommand{\Kt}{K_{t}}
\newcommand{\Lt}{L_{t}}

% --- Rates and prices ------------------------------------------------------
\newcommand{\inflation}{\pi_{t}}
\newcommand{\nomrate}{i_{t}}
\newcommand{\realrate}{r_{t}}
\newcommand{\growth}{g}
\newcommand{\deprec}{\delta}                    % depreciation rate
\newcommand{\discount}{\beta}                   % discount factor

\newcommand{\pc}[1]{\tilde{#1}}                 % per effective worker

\newcommand{\ss}[1]{\bar{#1}}                   % steady-state value

\newcommand{\CRRA}[2]{\frac{#1^{1-#2}-1}{1-#2}}
